  \node[rm io, minimum width=34mm] (in) at (0,0) {\texttt{Judgement} + \texttt{StoryRecord}};

  \node[rm dec] (d1b) at (3.6,-1.4) {story\\blocked\\before?};
  \node[rm box] (cont) at (7.5,-1.4) {keep blocking\\\tiny contain objects labeled since};
  \node[rm box] (nop) at (7.5,-2.35) {nothing\\\tiny the verdict is only logged};
  \node[rm dec] (d2) at (0,-3.1) {mode\\warn?};
  \node[rm box] (warn) at (4.2,-3.1) {notify admin once per higher decision\\\tiny ``would be held / blocked''; nothing paused};

  \node[rm block, align=center, font=\scriptsize\sffamily] (blk) at (4.2,-4.6)

  \node[rm dec] (d4) at (0,-6.2) {hold\\open?};
  \node[rm hold, align=center, font=\scriptsize\sffamily] (sh) at (4.3,-6.2)

     \tiny platform from device and roles; deadline = now + timeout;\\
     \tiny ``continue'' offered only if fewer than 3 questions failed};
  \node[rm human] (ep) at (9.9,-5.75) {endpoint: ask the user\\\tiny stop / continue; timeout = stop};
  \node[rm human] (sv) at (9.9,-6.75) {server: notify admin\\\tiny release / confirm; timeout = keep quarantined};
  \node[rm dec] (d5) at (0,-7.8) {hold\\pending?};
  \node[rm hold, align=center, font=\scriptsize\sffamily] (ext) at (4.2,-7.8)
    {\textbf{extend hold}: cover newly labeled objects\\\tiny all 3 questions failed meanwhile $\Rightarrow$ offer ``stop'' only};
  \node[rm term] (res) at (0,-9.1) {nothing (hold resolved)};

  \draw[rm flow] (in) -- (d1);

  \draw[rm flow] (d1) -- node[rm no, right] {no} (d2);
  \draw[rm flow] (d2) -- node[rm yes, above] {yes} (warn);
  \draw[rm flow] (d2) -- node[rm no, right] {no} (d3);
  \draw[rm flow] (d3) -- node[rm yes, above] {yes} (blk);
  \draw[rm flow] (d3) -- node[rm no, right] {no: HOLD} (d4);
